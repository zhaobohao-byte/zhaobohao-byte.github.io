# 进程处理
killall xxx # 杀死所有同名进程
ps aux | grep xxx # 查询符合名字的进程
kill PID	   # 杀死 PID 进程
kill -15 PID   # 正常终止（默认）
kill -9 PID    # 强制杀死
top # 系统自带进程查看
btop # 更美观，需要下载（实测多0.5M运行内存）

# 文件和目录操作
ls      # 列出目录内容，例如：ls -lah
cd      # 切换目录，例如：cd /var/log
pwd     # 显示当前工作目录，例如：pwd
mkdir   # 创建目录，例如：mkdir -p demo/test
rmdir   # 删除空目录，例如：rmdir empty_dir
touch   # 创建空文件或更新文件时间戳，例如：touch app.log
cp      # 复制文件或目录，例如：cp -r src backup
mv      # 移动或重命名文件/目录，例如：mv old.txt new.txt
rm      # 删除文件或目录，例如：rm -rf build
ln      # 创建硬链接或符号链接，例如：ln -s /opt/app app_link
# 文件查看和编辑
cat     # 查看文件内容（连接并显示），例如：cat /etc/hosts
tac     # 反向查看文件内容，例如：tac app.log
less    # 分页查看文件（可前后翻页），例如：less README.md
more    # 分页查看文件（只能向前），例如：more /etc/services
head    # 查看文件开头几行（默认10行），例如：head -n 20 app.log
tail    # 查看文件结尾几行（默认10行），例如：tail -f app.log
nano    # 简单文本编辑器，例如：nano notes.txt
vim     # 强大文本编辑器（或 vi），例如：vim ~/.bashrc
# 文件查找和搜索
find    # 在文件系统中搜索文件，例如：find . -name '*.py'
locate  # 快速查找文件（基于数据库），例如：locate libcudart.so
which   # 查找可执行文件位置，例如：which python3
whereis # 查找二进制、源代码和手册页位置，例如：whereis ssh
grep    # 在文件中搜索文本（支持正则），例如：grep -R 'TODO' .
# 权限管理
chmod   # 修改文件/目录权限，例如：chmod +x deploy.sh
chown   # 修改文件/目录所有者，例如：chown -R user:user data
chgrp   # 修改文件/目录所属组，例如：chgrp developers script.sh
umask   # 设置默认权限掩码，例如：umask 022
# 进程管理
ps      # 显示当前进程快照，例如：ps aux | grep python
top     # 实时查看进程（动态），例如：top
htop    # 更友好的交互式进程查看器（需安装），例如：htop
kill    # 向进程发送信号终止，例如：kill -9 12345
killall # 按名称终止进程，例如：killall chrome
pkill   # 按名称或条件终止进程，例如：pkill -f train.py
nice    # 以指定优先级启动进程，例如：nice -n 10 python3 task.py
renice  # 更改运行中进程的优先级，例如：renice 5 -p 12345
bg      # 将进程放到后台，例如：bg %1
fg      # 将后台进程调到前台，例如：fg %1
jobs    # 查看当前终端后台任务，例如：jobs
# 系统信息和监控
uname   # 显示系统信息，例如：uname -a
uptime  # 显示系统运行时间和负载，例如：uptime
df      # 查看磁盘空间使用情况，例如：df -h
du      # 查看目录或文件占用空间，例如：du -sh logs
free    # 查看内存使用情况，例如：free -h
lsblk   # 列出块设备（分区信息），例如：lsblk
dmesg   # 查看内核环缓冲区消息，例如：dmesg | tail
# 网络相关
ping    # 测试主机连通性，例如：ping -c 4 8.8.8.8
curl    # 发送HTTP请求/下载文件，例如：curl -I https://example.com
wget    # 非交互式下载文件，例如：wget https://example.com/file.tar.gz
ip      # 显示/配置网络接口（现代替代ifconfig），例如：ip addr
ss      # 查看网络连接（现代替代netstat），例如：ss -tulnp
nslookup / dig  # DNS查询，例如：dig openai.com
netstat # 查看网络状态（部分系统已弃用），例如：netstat -tulnp
# 压缩和归档
tar     # 创建/解压 tar 归档（常配合 gzip/bzip2/xz），例如：tar -czvf archive.tar.gz project/
gzip    # 压缩/解压 .gz 文件，例如：gzip app.log
gunzip  # 解压 .gz 文件，例如：gunzip app.log.gz
bzip2   # 压缩为 .bz2（更高压缩率），例如：bzip2 data.txt
zip     # 创建/解压 .zip 文件，例如：zip -r backup.zip docs/
unzip   # 解压 .zip 文件，例如：unzip backup.zip
# 用户和组管理
whoami  # 显示当前用户名，例如：whoami
id      # 显示用户和组信息，例如：id ubuntu
sudo    # 以超级用户权限执行命令，例如：sudo apt update
su      # 切换用户（默认root），例如：su - root
useradd # 添加用户，例如：sudo useradd -m devuser
passwd  # 修改用户密码，例如：passwd devuser
groupadd # 添加组，例如：sudo groupadd developers
# 包管理（常见发行版）
# Debian/Ubuntu 系
apt update          # 更新软件源缓存，例如：sudo apt update
apt upgrade         # 升级已安装软件包，例如：sudo apt upgrade
apt install <pkg>   # 安装软件包，例如：sudo apt install git
apt remove <pkg>    # 卸载软件包，例如：sudo apt remove vim
apt search <pkg>    # 搜索软件包，例如：apt search docker

# Red Hat/CentOS/Fedora 系（旧）
yum install <pkg>   # 安装软件包，例如：sudo yum install git
yum update          # 更新系统，例如：sudo yum update

# Fedora 现代 / RHEL 8+
dnf install <pkg>   # 安装软件包，例如：sudo dnf install git

# Arch Linux 系
pacman -S <pkg>     # 安装软件包，例如：sudo pacman -S git
pacman -Syu         # 同步并升级系统，例如：sudo pacman -Syu
# 其他常用工具
echo    # 输出文本，例如：echo 'hello world'
date    # 显示或设置系统时间，例如：date
history # 查看命令历史，例如：history | tail
alias   # 创建命令别名，例如：alias ll='ls -lah'
env     # 查看环境变量，例如：env | grep PATH
export  # 设置或导出环境变量，例如：export CUDA_HOME=/usr/local/cuda
shutdown # 关机或重启，例如：sudo shutdown -h now
reboot  # 重启系统，例如：sudo reboot
clear   # 清屏，例如：clear
man     # 查看命令手册，例如：man tar
info    # 查看命令信息（另一种手册格式），例如：info coreutils
# 命令行编辑/快捷键
Tab        # 自动补全命令、路径或文件名，例如：输入 cd Do 后按 Tab 补全为 Documents
Ctrl + A   # 光标移动到行首，例如：快速回到命令开头修改路径
Ctrl + E   # 光标移动到行尾，例如：在命令末尾追加参数
Ctrl + B   # 光标左移一个字符，例如：回退一位修改拼写
Ctrl + F   # 光标右移一个字符，例如：向前一位继续编辑
Ctrl + P   # 调出上一条历史命令，例如：快速重跑上一条 python 命令
Ctrl + N   # 调出下一条历史命令，例如：在历史记录中向后翻
Ctrl + U   # 删除光标前的全部内容，例如：一键清空已输入的前半行
Ctrl + K   # 删除光标后的全部内容，例如：保留前缀并删掉后半行
Ctrl + W   # 删除光标前一个单词，例如：删掉输错的参数
Alt + B    # 向后移动一个单词，例如：快速跳到上一个参数
Alt + F    # 向前移动一个单词，例如：快速跳到下一个参数
Alt + D    # 删除光标后的一个单词，例如：删掉后面的错误参数
Alt + .    # 插入上一条命令的最后一个参数，例如：复用刚刚使用过的文件路径
Ctrl + T   # 交换相邻两个字符，例如：把 sl 改正为 ls
Ctrl + _   # 撤销上一步编辑，例如：恢复刚才误删的字符
Ctrl + Y   # 粘贴最近删除的内容，例如：恢复刚用 Ctrl + U 或 Ctrl + K 删除的文本
Ctrl + R   # 搜索历史命令，例如：输入 ssh 查找之前执行过的 ssh 命令
Ctrl + L   # 清屏，例如：保留当前 shell 会话但清空显示内容
Ctrl + Z   # 挂起当前前台程序，例如：临时挂起 vim 回到 shell
Ctrl + D   # 发送 EOF 或退出当前 shell，例如：在空命令行下退出终端会话
