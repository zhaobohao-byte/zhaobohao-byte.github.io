pip list | grep 关键词## 清空 conda 源，恢复默认
conda config --remove-key default_channels
conda config --remove-key custom_channels
conda config --remove-key ssl_verify
# 或者直接删除配置文件
rm ~/.condarc

## pip 恢复默认
pip config unset global.index-url
pip config unset global.trusted-host

## 常用
conda env list
conda info --envs # 列出环境信息
conda env remove --name your_env_name # 删除环境
conda create -n <新环境名> --clone <旧环境名> # 复制环境
pip list | grep <关键词>
