2026年安装CLI，ubuntu

# 设置密钥环
curl -fsSL https://cli.github.com/packages/githubcli-archive-keyring.gpg | sudo dd of=/usr/share/keyrings/githubcli-archive-keyring.gpg
sudo chmod go+r /usr/share/keyrings/githubcli-archive-keyring.gpg
# 添加官方稳定版源
echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/githubcli-archive-keyring.gpg] https://cli.github.com/packages stable main" | sudo tee /etc/apt/sources.list.d/github-cli.list > /dev/null
# 安装
sudo apt update && sudo apt install gh -y
  • 使用
# 基础配置
gh auth login          # 交互式登录（支持浏览器或 Token）
gh auth status         # 查看当前登录状态及账号
gh auth logout         # 退出登录

# 仓库管理
gh repo create my-project --public --clone  # 创建新的公共仓库并克隆到本地
gh repo clone owner/repo                    # 克隆仓库
gh repo fork owner/repo --clone             # 派生仓库并克隆到本地
gh repo view --web                          # 直接在浏览器中打开当前仓库页面

# 拉取请求
gh pr create --title "修复Bug" --body "描述信息"  # 创建 PR
gh pr list                                        # 列出当前仓库所有的 PR
gh pr checkout 123                                # 检出 ID 为 123 的 PR 分支进行测试
gh pr diff                                        # 查看当前 PR 的代码改动
gh pr merge --merge                               # 合并 PR（支持 --rebase 或 --squash）
gh pr status                                      # 查看与你相关的 PR 状态
